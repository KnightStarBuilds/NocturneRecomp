# SDL 2.30.0 common code stays in the verified, build-local dependency copy.
# Only the three backend registries are extended; cached sources are immutable.
if(NOT NOCTURNE_PS4_SDL_SOURCE_DIR)
    message(FATAL_ERROR "NOCTURNE_PS4_SDL_SOURCE_DIR must name the prepared SDL2 source")
endif()
set(_sdl "${NOCTURNE_PS4_SDL_SOURCE_DIR}")
cmake_path(IS_PREFIX CMAKE_BINARY_DIR "${_sdl}" NORMALIZE _build_local)
if(NOT _build_local)
    message(FATAL_ERROR "PS4 SDL source must be a build-local copy, never deps/cache")
endif()

function(ps4_sdl_register relative anchor addition)
    set(_path "${_sdl}/${relative}")
    file(READ "${_path}" _text)
    string(FIND "${_text}" "${addition}" _already)
    if(_already EQUAL -1)
        string(FIND "${_text}" "${anchor}" _found)
        if(_found EQUAL -1)
            message(FATAL_ERROR "SDL 2.30.0 registration anchor missing: ${relative}")
        endif()
        string(REPLACE "${anchor}" "${addition}\n${anchor}" _text "${_text}")
        file(WRITE "${_path}" "${_text}")
    endif()
endfunction()
ps4_sdl_register(src/video/SDL_sysvideo.h "extern VideoBootStrap PSP_bootstrap;" "extern VideoBootStrap PS4_bootstrap;")
ps4_sdl_register(src/video/SDL_video.c "#ifdef SDL_VIDEO_DRIVER_PSP" "#ifdef SDL_VIDEO_DRIVER_PS4\n    &PS4_bootstrap,\n#endif")
ps4_sdl_register(src/audio/SDL_sysaudio.h "extern AudioBootStrap PSPAUDIO_bootstrap;" "extern AudioBootStrap PS4AUDIO_bootstrap;")
ps4_sdl_register(src/audio/SDL_audio.c "#ifdef SDL_AUDIO_DRIVER_PSP" "#ifdef SDL_AUDIO_DRIVER_PS4\n    &PS4AUDIO_bootstrap,\n#endif")
ps4_sdl_register(src/joystick/SDL_sysjoystick.h "extern SDL_JoystickDriver SDL_PSP_JoystickDriver;" "extern SDL_JoystickDriver SDL_PS4_JoystickDriver;")
ps4_sdl_register(src/joystick/SDL_joystick.c "#ifdef SDL_JOYSTICK_PSP" "#ifdef SDL_JOYSTICK_PS4\n    &SDL_PS4_JoystickDriver,\n#endif")

file(GLOB _headers "${_sdl}/include/*.h")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/include/SDL2")
foreach(_header IN LISTS _headers)
    get_filename_component(_name "${_header}" NAME)
    configure_file("${_header}" "${CMAKE_CURRENT_BINARY_DIR}/include/SDL2/${_name}" COPYONLY)
endforeach()
configure_file(SDL_config.h "${CMAKE_CURRENT_BINARY_DIR}/include/SDL2/SDL_config.h" COPYONLY)
set(_sources)
foreach(_dir "" atomic audio cpuinfo dynapi events file joystick haptic hidapi libm locale misc power render sensor stdlib thread timer video video/yuv2rgb render/software haptic/dummy sensor/dummy locale/dummy misc/dummy)
    file(GLOB _common "${_sdl}/src/${_dir}/*.c")
    list(APPEND _sources ${_common})
endforeach()
list(APPEND _sources "${_sdl}/src/thread/generic/SDL_syssem.c" "${_sdl}/src/thread/generic/SDL_systls.c")
list(REMOVE_DUPLICATES _sources)
add_library(SDL2-static STATIC ${_sources} video.c audio.c joystick.c system.c thread.c)
add_library(SDL2::SDL2 ALIAS SDL2-static)
add_library(SDL2::SDL2-static ALIAS SDL2-static)
target_include_directories(SDL2-static PUBLIC "${CMAKE_CURRENT_BINARY_DIR}/include/SDL2" "${CMAKE_CURRENT_BINARY_DIR}/include" PRIVATE "${_sdl}/src")
target_compile_definitions(SDL2-static PUBLIC SDL_BYTEORDER=1234 PRIVATE SDL_BUILDING_LIBRARY=1 DYNAPI_NEEDS_DLOPEN=1 USING_GENERATED_CONFIG_H=1 _GNU_SOURCE=1)
target_compile_options(SDL2-static PRIVATE -fno-strict-aliasing)
target_link_libraries(SDL2-static PUBLIC SceVideoOut SceAudioOut ScePad SceUserService SceSysmodule SceSystemService kernel)
set_target_properties(SDL2-static PROPERTIES OUTPUT_NAME SDL2 POSITION_INDEPENDENT_CODE ON)
